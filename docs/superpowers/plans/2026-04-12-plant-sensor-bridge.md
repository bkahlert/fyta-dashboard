# Plant Sensor Bridge Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create one HA template sensor per plant whose numeric state enables the moisture line graph while its `moisture_state` attribute enables exact categorical sort order; replace the broken firmware-update chip with a data-freshness chip.

**Architecture:** Three new/modified scripts. `generate-plant-sensors.mjs` discovers plant prefixes from HA and writes a YAML file the user places in their HA config. `reload-templates.mjs` fires the template reload service. `update-main-view.mjs` is updated to match `sensor.*_plant` entities via attribute-filtered include rules, restore the numeric sensor graph, and swap in the freshness chip.

**Tech Stack:** Node.js 22+ native WebSocket, HA WebSocket API (`get_states`, `call_service`, `lovelace/config/save`), HA template sensor YAML, `custom:auto-entities` attribute filter

**Token:** All scripts read `process.env.HA_TOKEN`.

## Task 1: Generate template sensor YAML

- [ ] **Step 1: Create `scripts/generate-plant-sensors.mjs`**

```js
import { writeFileSync, mkdirSync } from 'fs';
import { dirname } from 'path';

const TOKEN = process.env.HA_TOKEN;
if (!TOKEN) { console.error('HA_TOKEN not set'); process.exit(1); }

const outPath = process.argv[2] ?? 'config/fyta_plant_sensors.yaml';

function toTitleCase(s) {
  return s.replace(/_/g, ' ').replace(/\b\w/g, c => c.toUpperCase());
}

function generateYaml(prefixes) {
  const sensors = prefixes.map(p => [
    `      - name: "${toTitleCase(p)} Plant"`,
    `        unique_id: ${p}_plant`,
    `        state: "{{ states('sensor.${p}_moisture') | float(0) }}"`,
    `        unit_of_measurement: "%"`,
    `        device_class: moisture`,
    `        state_class: measurement`,
    `        attributes:`,
    `          moisture_state: "{{ states('sensor.${p}_moisture_state') }}"`,
  ].join('\n')).join('\n\n');

  return [
    '# FYTA plant sensor bridges — generated by scripts/generate-plant-sensors.mjs',
    '# Place in your HA config packages directory and reload template entities.',
    'template:',
    '  - sensor:',
    sensors,
    '',
  ].join('\n');
}

const ws = new WebSocket('ws://homeassistant.local:8123/api/websocket');
ws.onerror = (err) => { console.error('✘ WebSocket error:', err.message); process.exit(1); };
function send(obj) { ws.send(JSON.stringify(obj)); }
let msgId = 1;

ws.onmessage = (event) => {
  const msg = JSON.parse(event.data);
  if (msg.type === 'auth_required') {
    send({ type: 'auth', access_token: TOKEN });
  } else if (msg.type === 'auth_ok') {
    send({ id: msgId++, type: 'get_states' });
  } else if (msg.type === 'result') {
    const prefixes = msg.result
      .map(s => s.entity_id)
      .filter(id => /^sensor\.[a-z0-9_]+_moisture_state$/.test(id))
      .map(id => id.slice('sensor.'.length, -'_moisture_state'.length))
      .sort();

    if (prefixes.length === 0) {
      console.error('✘ No sensor.*_moisture_state entities found');
      ws.close(); process.exit(1);
    }

    const dir = dirname(outPath);
    if (dir && dir !== '.') mkdirSync(dir, { recursive: true });
    writeFileSync(outPath, generateYaml(prefixes), 'utf8');
    console.log(`✔ Found ${prefixes.length} plants: ${prefixes.join(', ')}`);
    console.log(`✔ Written to ${outPath}`);
    ws.close();
  }
};
ws.onclose = () => process.exit(0);
setTimeout(() => { console.error('✘ Timeout'); process.exit(1); }, 8000);
```

- [ ] **Step 2: Create `scripts/reload-templates.mjs`**

```js
const TOKEN = process.env.HA_TOKEN;
if (!TOKEN) { console.error('HA_TOKEN not set'); process.exit(1); }

const ws = new WebSocket('ws://homeassistant.local:8123/api/websocket');
ws.onerror = (err) => { console.error('✘ WebSocket error:', err.message); process.exit(1); };
function send(obj) { ws.send(JSON.stringify(obj)); }
let msgId = 1;

ws.onmessage = (event) => {
  const msg = JSON.parse(event.data);
  if (msg.type === 'auth_required') {
    send({ type: 'auth', access_token: TOKEN });
  } else if (msg.type === 'auth_ok') {
    send({
      id: msgId++,
      type: 'call_service',
      domain: 'homeassistant',
      service: 'reload_template_entities',
      service_data: {},
    });
  } else if (msg.type === 'result') {
    if (msg.success) {
      console.log('✔ Template entities reloaded');
    } else {
      console.error('✘ Reload failed:', JSON.stringify(msg.error));
      ws.close(); process.exit(1);
    }
    ws.close();
  }
};
ws.onclose = () => process.exit(0);
setTimeout(() => { console.error('✘ Timeout'); process.exit(1); }, 10000);
```
